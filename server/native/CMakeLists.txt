cmake_minimum_required(VERSION 3.20)
project(OpenGolfPhysics C)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(golf_physics_worker worker.c headless.c
    ${ROOT}/src/golf/physics.c ${ROOT}/src/golf/player.c
    ${ROOT}/src/common/bvh.c ${ROOT}/src/common/level.c ${ROOT}/src/common/maths.c
    ${ROOT}/src/common/vec.c ${ROOT}/src/common/map.c ${ROOT}/src/common/json.c ${ROOT}/src/common/base64.c
    ${ROOT}/src/3rd_party/parson/parson.c ${ROOT}/src/3rd_party/fast_obj/fast_obj.c)
target_include_directories(golf_physics_worker PRIVATE ${ROOT}/src ${ROOT}/src/3rd_party)
target_compile_definitions(golf_physics_worker PRIVATE GOLF_HEADLESS)
if(WIN32)
    target_link_libraries(golf_physics_worker PRIVATE psapi)
endif()
if(MSVC)
    target_compile_options(golf_physics_worker PRIVATE /Gy)
    target_link_options(golf_physics_worker PRIVATE /OPT:REF)
else()
    target_compile_options(golf_physics_worker PRIVATE -fms-extensions -ffunction-sections -fdata-sections)
    if(WIN32)
        target_link_options(golf_physics_worker PRIVATE -Wl,--gc-sections)
    elseif(APPLE)
        target_link_options(golf_physics_worker PRIVATE -Wl,-dead_strip)
    else()
        target_link_options(golf_physics_worker PRIVATE -Wl,--gc-sections)
        target_link_libraries(golf_physics_worker PRIVATE m)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    get_target_property(PHYSICS_SOURCES golf_physics_worker SOURCES)
    list(REMOVE_ITEM PHYSICS_SOURCES worker.c)
    add_executable(golf_physics_test physics_test.c ${PHYSICS_SOURCES})
    foreach(PROPERTY INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS LINK_OPTIONS LINK_LIBRARIES)
        get_target_property(VALUE golf_physics_worker ${PROPERTY})
        if(VALUE)
            set_target_properties(golf_physics_test PROPERTIES ${PROPERTY} "${VALUE}")
        endif()
    endforeach()
    add_test(NAME multiplayer_physics COMMAND golf_physics_test)
    set_tests_properties(multiplayer_physics PROPERTIES WORKING_DIRECTORY ${ROOT})
    foreach(LEVEL RANGE 1 20)
        add_test(NAME cup_capture_map_${LEVEL} COMMAND golf_physics_test "data/levels/level-${LEVEL}.level")
        set_tests_properties(cup_capture_map_${LEVEL} PROPERTIES WORKING_DIRECTORY ${ROOT})
    endforeach()
endif()
